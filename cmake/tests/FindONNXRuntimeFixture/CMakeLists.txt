cmake_minimum_required(VERSION 3.20)
project(FindONNXRuntimeFixture NONE)

if(NOT DEFINED FIND_ONNXRUNTIME_MODULE_DIR)
  message(FATAL_ERROR "FIND_ONNXRUNTIME_MODULE_DIR is required")
endif()
list(PREPEND CMAKE_MODULE_PATH "${FIND_ONNXRUNTIME_MODULE_DIR}")
find_package(ONNXRuntime REQUIRED MODULE)

if(NOT TARGET ONNXRuntime::ONNXRuntime)
  message(FATAL_ERROR "ONNXRuntime::ONNXRuntime was not exported")
endif()
get_target_property(_includes ONNXRuntime::ONNXRuntime INTERFACE_INCLUDE_DIRECTORIES)
if(NOT _includes MATCHES "[/\\]include$")
  message(FATAL_ERROR "ONNX Runtime include directory is invalid: ${_includes}")
endif()
if(WIN32)
  get_target_property(_implib ONNXRuntime::ONNXRuntime IMPORTED_IMPLIB)
  get_target_property(_runtime ONNXRuntime::ONNXRuntime IMPORTED_LOCATION)
  if(NOT _implib MATCHES "[/\\]lib[/\\]onnxruntime\\.lib$" OR
     NOT _runtime MATCHES "[/\\]lib[/\\]onnxruntime\\.dll$")
    message(FATAL_ERROR "ONNX Runtime Windows locations are invalid")
  endif()
endif()
