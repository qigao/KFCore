if(NOT WIN32 OR NOT KFCORE_ENABLE_ONNXRUNTIME)
  message(FATAL_ERROR "Hand preview requires Windows and ONNX Runtime")
endif()
if(NOT IS_DIRECTORY "$ENV{OPENCV_LITE_ROOT}")
  message(FATAL_ERROR "OPENCV_LITE_ROOT must point to the local opencv-lite SDK")
endif()
# This SDK has no package config and intentionally contains no videoio module.
file(STRINGS "$ENV{OPENCV_LITE_ROOT}/include/opencv2/core/version.hpp" _cv_versions
  REGEX "^#define CV_VERSION_(MAJOR|MINOR|REVISION)[ ]+[0-9]+")
set(_cv_suffix "")
foreach(_line IN LISTS _cv_versions)
  string(REGEX REPLACE ".*[ ]+([0-9]+)$" "\\1" _part "${_line}")
  string(APPEND _cv_suffix "${_part}")
endforeach()
foreach(_component core imgproc highgui imgcodecs)
  set(_library "$ENV{OPENCV_LITE_ROOT}/lib/opencv_${_component}${_cv_suffix}.lib")
  if(NOT EXISTS "${_library}")
    message(FATAL_ERROR "Missing opencv-lite library: ${_library}")
  endif()
  list(APPEND _preview_opencv_libraries "${_library}")
endforeach()
add_executable(kfhand_preview main.cpp experiment_dialog.cpp)
target_include_directories(kfhand_preview PRIVATE "$ENV{OPENCV_LITE_ROOT}/include")
target_link_libraries(kfhand_preview PRIVATE KFCore::mediapipe Salts::Capture KFCore::gesture_interaction ${_preview_opencv_libraries})
target_link_libraries(kfhand_preview PRIVATE comdlg32 gdiplus)
target_compile_options(kfhand_preview PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
set_target_properties(kfhand_preview PROPERTIES FOLDER "tools")
set(PREVIEW_RUNTIME_PATH "$ENV{PATH}")
configure_file(run.cmd.in "${CMAKE_BINARY_DIR}/bin/run-hand-preview.cmd" @ONLY)
